آموزش HTML

HTML برای تعریف ساختار و محتوای صفحات وب استفاده می‌شود.

درس ۱: ساختار اولیه HTML

هر صفحه HTML معمولاً با ساختار زیر شروع می‌شود.

<!DOCTYPE html>
<html lang="fa" dir="rtl">
<head>
  <meta charset="UTF-8">
  <title>صفحه من</title>
</head>
<body>
  <h1>سلام دنیا</h1>
  <p>این اولین صفحه من است.</p>
</body>
</html>

درس ۲: عنوان‌ها و پاراگراف‌ها

عنوان‌ها از h1 تا h6 هستند و p برای پاراگراف استفاده می‌شود.

<h1>عنوان اصلی</h1>
<h2>عنوان فرعی</h2>
<p>این یک پاراگراف است.</p>

درس ۳: لینک‌ها

برای پیوند به صفحه‌ای دیگر از عنصر a استفاده کن.

<a href="https://example.com">
  بازکردن سایت
</a>

درس ۴: تصاویر

ویژگی src آدرس تصویر و alt توضیح تصویر را مشخص می‌کند.

<img src="image.jpg" alt="توضیح تصویر">

درس ۵: فهرست‌ها

<ul>
  <li>HTML</li>
  <li>CSS</li>
  <li>JavaScript</li>
</ul>

درس ۶: فرم‌ها

<form>
  <label for="username">نام کاربری</label>
  <input id="username" name="username" type="text">
  <button type="submit">ارسال</button>
</form>

آموزش CSS

CSS ظاهر و چیدمان عناصر HTML را کنترل می‌کند.

درس ۱: انتخابگرها

p {
  color: blue;
}

.title {
  font-size: 28px;
}

#main {
  background-color: lightgray;
}

درس ۲: رنگ و پس‌زمینه

body {
  background-color: #f5f5f5;
  color: #222;
}

h1 {
  color: darkblue;
}

درس ۳: فاصله‌ها و کادرها

.card {
  margin: 20px;
  padding: 16px;
  border: 1px solid #ccc;
  border-radius: 12px;
}

درس ۴: چیدمان با Flexbox

.container {
  display: flex;
  justify-content: center;
  align-items: center;
  gap: 16px;
}

درس ۵: طراحی واکنش‌گرا

@media (max-width: 600px) {
  .container {
    flex-direction: column;
  }
}

آموزش JavaScript

JavaScript به صفحات وب قابلیت تعامل و واکنش به کاربر می‌دهد.

درس ۱: اولین برنامه

تابع console.log برای نمایش مقدار در کنسول مرورگر استفاده می‌شود.

console.log("سلام دنیا!");

درس ۲: متغیرها

از let برای متغیر قابل تغییر و از const برای انتسابی که دوباره انجام نمی‌شود استفاده کن.

let name = "سارا";
const age = 20;

console.log(name);
console.log(age);

درس ۳: انواع داده

const username = "Ali"; // رشته
const score = 18;        // عدد
const passed = true;     // بولی
const emptyValue = null;

درس ۴: عملگرها

const a = 10;
const b = 3;

console.log(a + b);
console.log(a - b);
console.log(a * b);
console.log(a / b);
console.log(a % b);

درس ۵: شرط‌ها

const score = 16;

if (score >= 10) {
  console.log("قبول شدی");
} else {
  console.log("نیاز به تمرین بیشتر داری");
}

درس ۶: حلقه‌ها

for (let i = 1; i <= 5; i++) {
  console.log(i);
}

درس ۷: توابع

function greet(name) {
  return "سلام " + name;
}

console.log(greet("سارا"));

درس ۸: آرایه‌ها

const fruits = ["سیب", "موز", "پرتقال"];

console.log(fruits[0]);

fruits.push("انگور");
console.log(fruits.length);

درس ۹: اشیاء

const student = {
  name: "سارا",
  age: 20
};

console.log(student.name);
console.log(student.age);

درس ۱۰: تغییر محتوای صفحه (DOM)

با انتخاب عناصر می‌توانیم محتوای صفحه را تغییر دهیم.

// این کد را پس از وجود عنصر در صفحه اجرا کن
const heading = document.querySelector("#demo-heading");

if (heading) {
  heading.textContent = "متن با JavaScript تغییر کرد";
}

درس ۱۱: رویداد کلیک

const button = document.querySelector("#demo-button");

if (button) {
  button.addEventListener("click", function () {
    alert("دکمه کلیک شد!");
  });
}

برای اجرای مثال، ابتدا یک دکمه با شناسه demo-button در HTML قرار بده.

درس ۱۲: تمرین JavaScript

یک متغیر به نام number بساز و با شرط بررسی کن که عدد مثبت است یا نه.

const number = 5;

if (number > 0) {
  console.log("عدد مثبت است");
} else if (number < 0) {
  console.log("عدد منفی است");
} else {
  console.log("عدد صفر است");
}
