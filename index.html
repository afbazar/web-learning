آموزش HTML

HTML زبان نشانه‌گذاری برای ساختاردهی محتوای صفحات وب است.

درس ۱: ساختار اصلی HTML

هر صفحه HTML معمولاً از بخش head و body تشکیل می‌شود.

<!DOCTYPE html>
<html lang="fa" dir="rtl">
<head>
  <meta charset="UTF-8">
  <title>صفحه من</title>
</head>
<body>
  <h1>سلام دنیا</h1>
  <p>اولین صفحه من</p>
</body>
</html>

درس ۲: عنوان‌ها و پاراگراف‌ها

عنوان‌ها از h1 تا h6 هستند. تگ p برای پاراگراف استفاده می‌شود.

<h1>عنوان اصلی</h1>
<h2>عنوان فرعی</h2>
<p>این یک پاراگراف است.</p>

درس ۳: قالب‌بندی متن

<strong>متن مهم</strong>
<em>متن تأکیدی</em>
<mark>متن هایلایت‌شده</mark>
<del>متن حذف‌شده</del>

درس ۴: لینک‌ها

<a href="https://example.com">
  بازکردن سایت
</a>

ویژگی href مقصد لینک را مشخص می‌کند.

درس ۵: تصاویر

<img src="images/photo.jpg"
     alt="توضیح تصویر"
     width="300">

ویژگی alt توضیحی متنی برای تصویر فراهم می‌کند.

درس ۶: فهرست‌ها

<ul>
  <li>HTML</li>
  <li>CSS</li>
</ul>

<ol>
  <li>درس اول</li>
  <li>درس دوم</li>
</ol>

درس ۷: جدول‌ها

<table>
  <tr>
    <th>نام</th>
    <th>سن</th>
  </tr>
  <tr>
    <td>سارا</td>
    <td>۲۰</td>
  </tr>
</table>

درس ۸: فرم‌ها

<form>
  <label for="username">نام کاربری</label>
  <input id="username" name="username" type="text">

  <button type="submit">ارسال</button>
</form>

درس ۹: عناصر معنایی

<header>سربرگ سایت</header>
<nav>منوی سایت</nav>
<main>محتوای اصلی</main>
<section>یک بخش</section>
<footer>پاورقی سایت</footer>

درس ۱۰: ویژگی‌های id و class

<h2 id="main-title">عنوان</h2>
<p class="description">توضیحات</p>

از id برای شناسهٔ یکتا و از class برای گروه‌بندی عناصر استفاده می‌شود.

آموزش CSS

CSS ظاهر و چیدمان عناصر HTML را مشخص می‌کند.

درس ۱: انتخابگرها

p {
  color: blue;
}

.title {
  font-size: 28px;
}

#main {
  background-color: lightgray;
}

درس ۲: رنگ و پس‌زمینه

body {
  color: #222;
  background-color: #f5f5f5;
}

h1 {
  color: royalblue;
}

درس ۳: فونت و اندازه متن

p {
  font-family: sans-serif;
  font-size: 18px;
  line-height: 1.8;
}

درس ۴: فاصله‌ها و کادرها

.card {
  margin: 20px;
  padding: 16px;
  border: 1px solid #ccc;
  border-radius: 12px;
}

درس ۵: Flexbox

.container {
  display: flex;
  justify-content: center;
  align-items: center;
  gap: 16px;
}

درس ۶: طراحی واکنش‌گرا

@media (max-width: 600px) {
  .container {
    flex-direction: column;
  }
}

آموزش JavaScript

JavaScript به سایت امکان می‌دهد به کلیک‌ها و ورودی‌های کاربر واکنش نشان دهد.

درس ۱: اولین برنامه

console.log("سلام دنیا!");

این دستور پیام را در کنسول مرورگر نمایش می‌دهد.

درس ۲: متغیرها

let username = "سارا";
const age = 20;

console.log(username);
console.log(age);

let برای متغیر قابل‌تغییر و const برای مقداری است که دوباره به آن متغیر انتساب داده نمی‌شود.

درس ۳: انواع داده

const name = "علی"; // رشته
const age = 18;     // عدد
const passed = true; // بولی
const result = null;

درس ۴: عملگرها

const a = 10;
const b = 3;

console.log(a + b);
console.log(a - b);
console.log(a * b);
console.log(a / b);
console.log(a % b);

درس ۵: شرط‌ها

const score = 16;

if (score >= 10) {
  console.log("قبول شدی");
} else {
  console.log("نیاز به تمرین داری");
}

درس ۶: حلقه‌ها

for (let i = 1; i <= 5; i++) {
  console.log(i);
}

درس ۷: توابع

function greet(name) {
  return "سلام " + name;
}

console.log(greet("سارا"));

درس ۸: آرایه‌ها

const fruits = ["سیب", "موز", "پرتقال"];

console.log(fruits[0]);

fruits.push("انگور");
console.log(fruits.length);

درس ۹: اشیاء

const student = {
  name: "سارا",
  age: 20
};

console.log(student.name);

درس ۱۰: دست‌کاری DOM

با DOM می‌توان عناصر صفحه را انتخاب و محتوای آن‌ها را تغییر داد.

const heading = document.querySelector("#demo-heading");

if (heading) {
  heading.textContent = "عنوان تغییر کرد";
}

درس ۱۱: رویداد کلیک

const button = document.querySelector("#demo-button");

if (button) {
  button.addEventListener("click", function () {
    alert("دکمه کلیک شد!");
  });
}

درس ۱۲: توابع پیکانی

const add = (a, b) => a + b;

console.log(add(4, 6));

درس ۱۳: تمرین

برنامه‌ای بنویس که بررسی کند عدد مثبت است، منفی است یا صفر.

const number = 5;

if (number > 0) {
  console.log("مثبت");
} else if (number < 0) {
  console.log("منفی");
} else {
  console.log("صفر");
}
